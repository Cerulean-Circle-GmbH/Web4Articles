/**
 * Rename PDCA file using standardized naming patterns
 * Supports multiple rename strategies: now, creationDate, strip, feature
 * Automatically updates all dual link references
 * 
 * @param filePath Path to PDCA file to rename
 * @param renameCase Rename strategy to apply
 * @param dryRun If 'true', show plan without executing
 * @cliSyntax filePath renameCase dryRun
 * @cliDefault dryRun "false"
 * @cliValues renameCase now creationDate strip feature
 * @cliValues dryRun true false
 */
async rename(filePath: string, renameCase: string, dryRun: string = 'false'): Promise<this> {
  const fs = await import('fs');
  const path = await import('path');
  
  const projectRoot = await this.getProjectRoot();
  const normalized = this.normalizePathInternal(filePath, projectRoot, path);
  const fullPath = path.join(projectRoot, normalized);
  
  // Validation
  if (!fs.existsSync(fullPath)) {
    console.error(`❌ Error: File not found: ${normalized}`);
    return this;
  }
  
  if (!fullPath.endsWith('.pdca.md')) {
    console.error(`❌ Error: Not a PDCA file: ${normalized}`);
    console.log(`💡 rename only works with .pdca.md files\n`);
    return this;
  }
  
  const validCases = ['now', 'creationDate', 'strip', 'feature'];
  if (!validCases.includes(renameCase)) {
    console.error(`❌ Error: Invalid rename case: ${renameCase}`);
    console.log(`💡 Valid cases: ${validCases.join(', ')}\n`);
    return this;
  }
  
  // Calculate new name based on case
  const newName = await this.calculateRenameInternal(fullPath, renameCase, projectRoot, path, fs);
  
  if (!newName) {
    console.log(`ℹ️  No rename needed - file already matches target format\n`);
    return this;
  }
  
  const newPath = path.join(path.dirname(normalized), newName);
  
  // Display plan
  console.log(`\n📝 Rename Plan (${renameCase})\n`);
  console.log(`📂 From: ${path.basename(normalized)}`);
  console.log(`📂 To:   ${newName}\n`);
  
  if (dryRun === 'true') {
    console.log(`💡 Run without 'true' to execute: pdca rename "${filePath}" ${renameCase}`);
    return this;
  }
  
  // Delegate to mv for actual rename + link updates
  return await this.mv(normalized, newPath, dryRun);
}

/**
 * Generate new filename using git creation timestamp
 * @cliHide
 */
private async renameToCreationDateInternal(
  fullPath: string,
  filename: string,
  projectRoot: string
): Promise<string | null> {
  const { execSync } = await import('child_process');
  
  try {
    // Get creation date using git log --follow (handles renames)
    const gitDate = execSync(
      `git log --follow --format=%aI --reverse "${fullPath}" | head -1`,
      { cwd: projectRoot, encoding: 'utf-8' }
    ).trim();
    
    if (!gitDate) {
      console.error(`❌ Error: File not in git history`);
      return null;
    }
    
    // Parse ISO 8601 format: 2025-10-31T11:13:09+00:00
    const date = new Date(gitDate);
    const year = date.getUTCFullYear();
    const month = String(date.getUTCMonth() + 1).padStart(2, '0');
    const day = String(date.getUTCDate()).padStart(2, '0');
    const hour = String(date.getUTCHours()).padStart(2, '0');
    const minute = String(date.getUTCMinutes()).padStart(2, '0');
    
    const newTimestamp = `${year}-${month}-${day}-UTC-${hour}${minute}`;
    
    // Preserve feature flag if present
    const hasFeature = filename.includes('.feature.');
    return hasFeature ? `${newTimestamp}.feature.pdca.md` : `${newTimestamp}.pdca.md`;
    
  } catch (error: any) {
    console.error(`❌ Error extracting git creation date: ${error.message}`);
    return null;
  }
}
