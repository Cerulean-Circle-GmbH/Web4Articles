```typescript
let idealminimalcomponentCLI: string;

beforeAll(async () => {
  // Check if IdealMinimalComponent exists in test isolation
  const testDataPath = join(componentRoot, 'test/data/components/IdealMinimalComponent');
  
  if (existsSync(testDataPath)) {
    // Use test isolation version
    const versions = readdirSync(testDataPath).filter(v => /^\d+\.\d+\.\d+\.\d+$/.test(v));
    if (versions.length > 0) {
      const version = versions[0]; // Use first available version
      idealminimalcomponentCLI = join(testDataPath, version, 'idealminimalcomponent');
      return;
    }
  }
  
  // Fallback: Create IdealMinimalComponent 0.1.0.0 for this test
  const web4ts = new DefaultWeb4TSComponent().init({ projectRoot });
  await web4ts.create({ component: 'IdealMinimalComponent', version: '0.1.0.0' });
  
  idealminimalcomponentCLI = join(projectRoot, 'components/IdealMinimalComponent/0.1.0.0/idealminimalcomponent');
});
```
