**3. IMPLEMENT: DefaultWeb4TSComponent.ts**
```typescript
import { MethodSignature } from '../layer3/MethodSignature.interface.js';

export class DefaultWeb4TSComponent implements Web4TSComponent {
  model!: Web4TSComponentModel;
  
  /**
   * Component's method signatures (class metadata, not model state)
   * @pdca 2025-11-05-UTC-1158.pdca.md - Self-discovery in init()
   */
  private methods: Map<string, MethodSignature> = new Map();
  
  /**
   * Discover this component's methods from its prototype chain
   * Called automatically in init() - component discovers ITSELF
   */
  private discoverMethods(): void {
    const prototype = Object.getPrototypeOf(this);
    const methodNames = Object.getOwnPropertyNames(prototype)
      .filter((name) => typeof prototype[name] === "function")
      .filter((name) => !name.startsWith("_") && name !== "constructor");

    for (const methodName of methodNames) {
      const method = prototype[methodName];
      this.methods.set(methodName, {
        name: methodName,
        paramCount: method.length,
        isAsync: method.constructor.name === "AsyncFunction",
      });
    }
  }
  
  /**
   * Check if component has a method (Component interface)
   */
  hasMethod(name: string): boolean {
    return this.methods.has(name);
  }
  
  /**
   * Get method signature (Component interface)
   */
  getMethodSignature(name: string): MethodSignature | null {
    return this.methods.get(name) || null;
  }
  
  /**
   * List all method names (Component interface)
   */
  listMethods(): string[] {
    return Array.from(this.methods.keys());
  }
  
  /**
   * Initialize component - Radical OOP pattern
   */
  init(scenario?: Scenario<Web4TSComponentModel>): this {
    // ... existing init code ...
    
    // ✅ Self-discovery: Component discovers ITSELF when created
    // @pdca 2025-11-05-UTC-1158.pdca.md - Event-driven, not batch
    this.discoverMethods();
    
    return this;
  }
}
```

**4. UPDATE: DefaultCLI.ts**
```typescript
import { Reference } from '../layer3/Reference.interface.js';
import { Component } from '../layer3/Component.interface.js';
import { MethodSignature } from '../layer3/MethodSignature.interface.js';

export abstract class DefaultCLI implements CLI, Component<CLIModel> {
  model!: CLIModel;
  
  /**
   * CLI's own method signatures (class metadata)
   * @pdca 2025-11-05-UTC-1158.pdca.md - Renamed from methodSignatures
   */
  private cliMethods: Map<string, MethodSignature> = new Map();
  
  /**
   * Component references using Reference<T> (nullable by design)
   * @pdca 2025-11-05-UTC-1158.pdca.md - Changed from Component? to Reference<Component>
   */
  protected component: Reference<Component> = null;
  protected context: Reference<Component> = null;
  
  /**
   * Discover CLI's own methods (NOT component or context!)
   */
  private discoverMethods(): void {
    let currentPrototype = Object.getPrototypeOf(this);
    while (currentPrototype && currentPrototype !== Object.prototype) {
      const methodNames = Object.getOwnPropertyNames(currentPrototype)
        .filter((name) => typeof currentPrototype[name] === "function")
        .filter((name) => !name.startsWith("_") && name !== "constructor");

      for (const methodName of methodNames) {
        if (!this.cliMethods.has(methodName)) {
          const method = currentPrototype[methodName];
          this.cliMethods.set(methodName, {
            name: methodName,
            paramCount: method.length,
            isAsync: method.constructor.name === "AsyncFunction",
          });
        }
      }
      currentPrototype = Object.getPrototypeOf(currentPrototype);
    }
  }
  
  /**
   * Initialize CLI
   */
  init(scenario?: Scenario<CLIModel>): this {
    // ... existing init code ...
    
    // ✅ CLI discovers ITSELF
    this.discoverMethods();
    
    return this;
  }
  
  /**
   * Execute method with type-safe component interface
   */
  protected async tryExecuteMethod(command: string, args: string[]): Promise<boolean> {
    let signature: MethodSignature | null = null;
    let target: any;
    
    // Priority 1: CLI methods
    if (this.cliMethods.has(command)) {
      signature = this.cliMethods.get(command)!;
      target = this;
    }
    // Priority 2: Context methods (type-safe!)
    else if (this.context !== null && this.context.hasMethod(command)) {
      signature = this.context.getMethodSignature(command);
      target = this.context;
    }
    // Priority 3: Component methods (type-safe!)
    else if (this.component !== null && this.component.hasMethod(command)) {
      signature = this.component.getMethodSignature(command);
      target = this.component;
    }
    
    if (signature === null) return false;
    
    // ... validation and execution ...
    if (signature.isAsync) {
      await target[command](...args);
    } else {
      target[command](...args);
    }
    
    return true;
  }
  
  /**
   * Component interface implementation for CLI
   */
  hasMethod(name: string): boolean {
    return this.cliMethods.has(name);
  }
  
  getMethodSignature(name: string): MethodSignature | null {
    return this.cliMethods.get(name) || null;
  }
  
  listMethods(): string[] {
    return Array.from(this.cliMethods.keys());
  }
}
```
