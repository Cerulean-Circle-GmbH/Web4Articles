    return `${this.colors.promptWhite}your ${this.colors.promptCyan}web4${this.colors.promptWhite} command >${this.colors.reset} ${coloredCommand}`;
  }
  
  private buildColoredCommand(words: string[]): string {
    // Clean OOP method for command coloring
  }
}

// 2. CompletionFormatter class - dedicated to DISPLAY/WORD protocol
class CompletionFormatter {
  constructor(private promptConstructor: PromptConstructor) {}
  
  formatWithPrompt(values: string[]): string[] {
    const lines: string[] = [];
    
    // Add completion values
    values.forEach(value => lines.push(`DISPLAY: ${value}`));
    
    // Add prompt (NO EMPTY LINE!)
    const prompt = this.promptConstructor.buildShellPrompt();
    lines.push(`DISPLAY: ${prompt}`);
    
    // Add WORD lines
    this.addWordLines(lines, values);
    
    return lines;
  }
}

class DefaultCLI {
