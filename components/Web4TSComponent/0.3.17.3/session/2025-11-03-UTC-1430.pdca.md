## Objective

**Problem**: Completion behavior differs based on which `source.env` is sourced, making it hard to verify correct operation and causing confusion during development.

**Solution**: Define clear test scenarios that should ALL produce identical completion results, ensuring environment-agnostic behavior.

**Feature Request**: Add `web4tscomponent initProject force` to fully overwrite project files when needed.

### 2. Implement `initProject force` Feature

**Current Behavior** (`web4tscomponent initProject`):
- Smart timestamp-based sync
- Only updates if template is newer than existing file
- Safe but can't force updates

**New Behavior** (`web4tscomponent initProject force`):
- Unconditionally overwrites ALL project files
- Useful when:
  - Template timestamp is wrong
  - Manual edits need to be reverted
  - Testing clean installation
  - Resolving environment conflicts

**Files to Overwrite**:
- `tsconfig.json`
- `package.json`
- `source.env`

### 3. Create Automated Test Suite

Create `test/vitest/environment-completion-consistency.test.ts`:
- Test all 4 scenarios programmatically
- Verify identical output
- Document expected behavior
- Run in CI/CD

async initProject(force?: boolean): Promise<this> {
  const projectRoot = this.model.targetDirectory || process.cwd();
  console.log(`🚀 Initializing Web4 project at: ${projectRoot}`);
  
  const currentDir = path.dirname(new URL(import.meta.url).pathname);
  const fs = await import('fs/promises');
  
  // Update tsconfig.json
  const tsconfigPath = path.join(projectRoot, 'tsconfig.json');
  if (existsSync(tsconfigPath) && !force) {
    // Smart sync: check timestamps
    const tsconfigStats = await fs.stat(tsconfigPath);
    const templateStats = await fs.stat(tsconfigTemplatePath);
    
    if (templateStats.mtime > tsconfigStats.mtime) {
      // Update...
    } else {
      console.log(`   ℹ️  tsconfig.json already up to date`);
    }
  } else {
    // Force mode OR file doesn't exist - always write
    if (force && existsSync(tsconfigPath)) {
      console.log(`   🔨 Force updating tsconfig.json`);
    }
    const tsconfigContent = await this.loadTemplate('config/root-tsconfig.json.template', {});
    await fs.writeFile(tsconfigPath, tsconfigContent);
    console.log(`   ✅ ${force ? 'Force updated' : 'Created'} tsconfig.json`);
  }
  
  // Same logic for package.json and source.env
  // ...
  
  return this;
}

### Step 3: Create Test Suite

**File**: `components/Web4TSComponent/0.3.17.3/test/vitest/environment-completion-consistency.test.ts`

```typescript
import { describe, it, expect, beforeAll } from 'vitest';
import { execSync } from 'child_process';
import { readFileSync, existsSync } from 'fs';
import { join } from 'path';

describe('Environment Completion Consistency', () => {
  const projectRoot = process.cwd();
  const testCommand = 'idealminimalcomponent complete \'{"ior":{"uuid":"test","component":"IdealMinimalComponent","version":"0.1.0.0"},"completionCompWords":["idealminimalcomponent","li"],"completionCompCword":1}\'';
  
  // Expected output signature (all scenarios must match)
  const expectedSignature = [
    'DISPLAY: links',
    'DISPLAY: ────────────────────────────────────────────────────────────',
    'DISPLAY: 📖 Documentation:',
    'DISPLAY: Show semantic version links',
    'DISPLAY: your web4 command >',
    'WORD: links'
  ];
  
  const extractSignature = (output: string): string[] => {
    return output.split('\n')
      .filter(line => line.startsWith('DISPLAY:') || line.startsWith('WORD:'))
      .map(line => {
        // Strip ANSI codes for comparison
        return line.replace(/\x1b\[[0-9;]*m/g, '');
      })
      .filter(line => {
        // Include key lines only
        return line.includes('links') || 
               line.includes('────') || 
               line.includes('Documentation') || 
               line.includes('Show semantic') ||
               line.includes('your web4 command') ||
               line.startsWith('WORD:');
      });
  };
  
  it('Scenario 1: Production Component Environment', () => {
    const componentPath = join(projectRoot, 'components/IdealMinimalComponent/0.1.0.0');
    
    if (!existsSync(componentPath)) {
      console.log('⏩ Skipping: IdealMinimalComponent not created yet');
      return;
    }
    
    const output = execSync(testCommand, {
      cwd: componentPath,
      encoding: 'utf-8'
    });
    
    const signature = extractSignature(output);
    expect(signature.length).toBeGreaterThan(0);
    expect(signature).toContain('WORD: links');
    expect(output).toContain('your web4 command >');
  });
  
  it('Scenario 2: Test Isolation Environment', () => {
    const testIsolationPath = join(projectRoot, 'components/Web4TSComponent/0.3.17.3/test/data/components/IdealMinimalComponent/0.1.0.0');
    
    if (!existsSync(testIsolationPath)) {
      console.log('⏩ Skipping: Test isolation component not created yet');
      return;
    }
    
    const output = execSync(testCommand, {
      cwd: testIsolationPath,
      encoding: 'utf-8'
    });
    
    const signature = extractSignature(output);
    expect(signature.length).toBeGreaterThan(0);
    expect(signature).toContain('WORD: links');
    expect(output).toContain('your web4 command >');
  });
  
  it('Scenario 3: Web4TSComponent Development Environment', () => {
    const web4tsPath = join(projectRoot, 'components/Web4TSComponent/0.3.17.3');
    
    const output = execSync(testCommand, {
      cwd: web4tsPath,
      encoding: 'utf-8'
    });
    
    const signature = extractSignature(output);
    expect(signature.length).toBeGreaterThan(0);
    expect(signature).toContain('WORD: links');
    expect(output).toContain('your web4 command >');
  });
  
  it('Scenario 4: Project Root Environment', () => {
    const output = execSync(testCommand, {
      cwd: projectRoot,
      encoding: 'utf-8'
    });
    
    const signature = extractSignature(output);
    expect(signature.length).toBeGreaterThan(0);
    expect(signature).toContain('WORD: links');
    expect(output).toContain('your web4 command >');
  });
  
  it('All scenarios produce identical signatures', () => {
    const scenarios = [
      { name: 'Production', path: 'components/IdealMinimalComponent/0.1.0.0' },
      { name: 'Test Isolation', path: 'components/Web4TSComponent/0.3.17.3/test/data/components/IdealMinimalComponent/0.1.0.0' },
      { name: 'Web4TS Dev', path: 'components/Web4TSComponent/0.3.17.3' },
      { name: 'Project Root', path: '.' }
    ];
    
    const signatures: Record<string, string[]> = {};
    
    for (const scenario of scenarios) {
      const scenarioPath = join(projectRoot, scenario.path);
      
      if (!existsSync(scenarioPath)) {
        console.log(`⏩ Skipping ${scenario.name}: Path doesn't exist`);
        continue;
      }
      
      try {
        const output = execSync(testCommand, {
          cwd: scenarioPath,
          encoding: 'utf-8'
        });
        
        signatures[scenario.name] = extractSignature(output);
      } catch (error) {
        console.error(`❌ ${scenario.name} failed:`, (error as Error).message);
      }
    }
    
    // Compare all signatures
    const signatureKeys = Object.keys(signatures);
    if (signatureKeys.length > 1) {
      const baseline = signatures[signatureKeys[0]];
      
      for (let i = 1; i < signatureKeys.length; i++) {
        const current = signatures[signatureKeys[i]];
        expect(current).toEqual(baseline);
      }
      
      console.log(`✅ All ${signatureKeys.length} scenarios produce identical output`);
    }
  });
});
```
