```bash
# ============================================================================
# 🎯 Tab Completion Setup for Web4 CLIs
# ============================================================================

# Load bash-completion framework if available
_web4_load_bash_completion() {
    type _init_completion &>/dev/null && return 0
    local paths=(
        "/usr/share/bash-completion/bash_completion"
        "/etc/bash_completion"
        "/opt/homebrew/etc/profile.d/bash_completion.sh"
        "/usr/local/etc/bash_completion"
        "/usr/local/etc/profile.d/bash_completion.sh"
    )
    for p in "${paths[@]}"; do
        [ -f "$p" ] && . "$p" && return 0
    done
    return 1
}

# Generic TSCompletion-based completion function
_web4_tscompletion() {
    local component="$1" cli="$2"
    local cur="${COMP_WORDS[COMP_CWORD]}" args=("${COMP_WORDS[@]:1}")
    local tsc="$WEB4_PROJECT_ROOT/components/$component/latest/src/ts/layer4/TSCompletion.ts"
    [ ! -f "$tsc" ] && return 0
    local out=$(NODE_NO_WARNINGS=1 node --loader ts-node/esm "$tsc" "${args[@]}" 2>/dev/null || true)
    COMPREPLY=( $(compgen -W "$out" -- "$cur") )
    compopt -o default 2>/dev/null || true
}

# Auto-register completions for all CLIs
_web4_register_completions() {
    local scripts="$WEB4_PROJECT_ROOT/scripts" registered=()
    [ ! -d "$scripts" ] && return 0
    
    for cli in "$scripts"/*; do
        [ -x "$cli" ] && [ ! -d "$cli" ] || continue
        local name=$(basename "$cli")
        [[ "$name" =~ -v[0-9]+\.[0-9]+\.[0-9]+\.[0-9]+ ]] && continue
        
        if [ -L "$cli" ]; then
            local target=$(readlink "$cli")
            if [[ "$target" =~ components/([^/]+)/latest ]]; then
                local comp="${BASH_REMATCH[1]}"
                local tsc="$WEB4_PROJECT_ROOT/components/$comp/latest/src/ts/layer4/TSCompletion.ts"
                if [ -f "$tsc" ]; then
                    eval "_${name}_completion() { _web4_tscompletion '$comp' '$name'; }"
                    complete -F "_${name}_completion" "$name"
                    registered+=("$name")
                fi
            fi
        fi
    done
    
    [ ${#registered[@]} -gt 0 ] && echo "🎯 Tab completion enabled for: ${registered[*]}"
}

# Initialize completions
_web4_load_bash_completion >/dev/null 2>&1
_web4_register_completions
```

**What This Does:**
- Loads bash-completion framework (cross-platform paths)
- Defines `_web4_tscompletion()` function that calls TSCompletion.ts
- Auto-discovers all CLI scripts in `scripts/` directory
- Registers bash completion function for each discovered CLI
- Reports which CLIs have completion enabled

**Test 2-6: TSCompletion with method/parameter names** ⚠️

**Command tested:**
```bash
NODE_NO_WARNINGS=1 node --loader ts-node/esm components/Web4TSComponent/0.3.9.1/src/ts/layer4/TSCompletion.ts "links"
```
**Result:** Empty output (no completions returned)

**Root Cause Analysis:**

1. **TSCompletion expects 2 args:** `[className, methodName]`
   - Example: `tssh TSsh methodName`
   - This is the `tssh` pattern from TSRanger

2. **web4tscomponent CLI has NO className in user input**
   - User types: `web4tscomponent links`
   - Args passed to TSCompletion: `["links"]` (only 1 arg)
   - TSCompletion doesn't know what class to inspect!

3. **The gap:** Shell → TypeScript handoff missing class context

**Critical Insight: Eliminate tssh from Design**

- **NOT DESIRED:** Using `tssh` CLI pattern (className in user input)
- **DESIRED:** Direct `web4tscomponent methodName` pattern
- **REQUIREMENT:** Shell completion script must inject class name BEFORE calling TypeScript
- **ARCHITECTURE:** 
  - Shell script: MINIMAL (only <Tab> → TypeScript handoff)
  - TypeScript: ALL LOGIC (TSCompletion + DefaultCLI + Web4TSComponentCLI)
  - TSCompletion must know exact CLI class instance to inspect

**Decision: Fix bash function to inject class name**

**WHY:**
- Keeps shell logic minimal (just add className)
- All real logic stays in TypeScript (TSCompletion + CLI classes)
- TSCompletion already has all infrastructure, just needs correct args
- User experience unchanged: `web4tscomponent links` (no className visible)

**HOW:**
```bash
_web4_tscompletion() {
    local component="$1" cli="$2"
    local cur="${COMP_WORDS[COMP_CWORD]}" args=("${COMP_WORDS[@]:1}")
    local tsc="$WEB4_PROJECT_ROOT/components/$component/latest/src/ts/layer4/TSCompletion.ts"
    [ ! -f "$tsc" ] && return 0
    
    # INJECT CLI class name as first arg for TSCompletion
    local cliClassName="${component}CLI"  # e.g., Web4TSComponentCLI
    local completeArgs=("$cliClassName" "${args[@]}")
    
    local out=$(NODE_NO_WARNINGS=1 node --loader ts-node/esm "$tsc" "${completeArgs[@]}" 2>/dev/null || true)
    COMPREPLY=( $(compgen -W "$out" -- "$cur") )
    compopt -o default 2>/dev/null || true
}
```

**What This Achieves:**
- User types: `web4tscomponent links <Tab>`
- Bash passes to TSCompletion: `["Web4TSComponentCLI", "links"]`
- TSCompletion knows to inspect `Web4TSComponentCLI` class
- TSCompletion returns parameter values: `['', 'fix', 'verify', 'show', 'list']`
- User sees completions, types `f<Tab>`, gets `fix`
