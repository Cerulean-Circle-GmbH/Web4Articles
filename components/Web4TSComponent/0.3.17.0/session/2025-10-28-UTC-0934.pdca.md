### **Init() Pattern Refactoring Plan**

> **User Feedback (2025-10-28):**
> "in DefaultCLI createEmptyModel does the job that normally is to be done by init. The default Web4 instantiation pattern is `let component = new DefaultComponent().init();` perfect would be even `let component = DefaultComponent.new.init();` for ALL Web4 components. IF the init has the optional parameter, it initializes the model from the scenario instead."

**Current Problem:**
- `createEmptyModel()` is called in `constructor()` - violates Web4 pattern
- `init()` only merges scenario data, doesn't create empty model
- Can't do `new DefaultComponent()` without initialization

**Target Pattern:**
```typescript
// Empty initialization

// Future: Static factory
let component = DefaultComponent.new.init();

// With scenario

```

```typescript
export interface Component<TModel extends Model = Model> {
```

```typescript
}
```

**3. Create DefaultVersion Class (layer2/DefaultVersion.ts)**
```typescript
/**
 * DefaultVersion - Web4 Component for version management
 * Web4 pattern: Empty constructor + init(scenario) + toScenario()
 * 
 * Simplicity: No VersionParts type, no VersionPromotion enum
 * Methods are self-explanatory: promoteMajor(), promoteMinor(), etc.
 */

import { Version } from '../layer3/Version.interface.js';
import { VersionModel } from '../layer3/VersionModel.interface.js';
import { Scenario } from '../layer3/Scenario.interface.js';
import { randomUUID } from 'crypto';

export class DefaultVersion implements Version {
  private model!: VersionModel;
  
  /**
   * Empty constructor - Web4 radical OOP pattern
   */
  constructor() {
    // Empty - all initialization in init()
  }
  
  /**
   * Initialize with scenario
   */
  init(scenario?: Scenario<VersionModel>): this {
    if (!this.model) {
      this.model = {
        uuid: randomUUID(),
        name: 'version',
        origin: 'system',
        definition: 'Version component',
        major: 0,
        minor: 0,
        patch: 0,
        revision: 0,
        versionString: '0.0.0.0'
      };
    }
    
    if (scenario?.model) {
      this.model = { ...this.model, ...scenario.model };
    }
    
    return this;
  }
  
  /**
   * Convert to Scenario for persistence
   */
  async toScenario(name?: string): Promise<Scenario<VersionModel>> {
    const ownerData = JSON.stringify({
      user: process.env.USER || 'system',
      hostname: process.env.HOSTNAME || 'localhost',
      uuid: this.model.uuid,
      timestamp: new Date().toISOString(),
      component: 'Version',
      version: this.model.versionString
    });
    
    return {
      ior: {
        uuid: this.model.uuid,
        component: 'Version',
        version: this.model.versionString
      },
      owner: ownerData,
      model: this.model
    };
  }
  
  /**
   * Static factory: Create from version string
   */
  static fromString(version: string): DefaultVersion {
    return new DefaultVersion().init().parse(version);
  }
  
  /**
   * Parse version string and update model
   */
  parse(version: string): this {
    const pattern = /^(\d+)\.(\d+)\.(\d+)\.(\d+)$/;
    const match = version.match(pattern);
    
    if (!match) {
      throw new Error(`Invalid version format: ${version}. Expected X.Y.Z.W`);
    }
    
    const [, major, minor, patch, revision] = match.map(Number);
    
    this.model.major = major;
    this.model.minor = minor;
    this.model.patch = patch;
    this.model.revision = revision;
    this.model.versionString = version;
    
    return this;
  }
  
  /**
   * Validate version string format
   */
  isValid(version: string): boolean {
    return /^\d+\.\d+\.\d+\.\d+$/.test(version);
  }
  
  /**
   * Convert to string
   */
  toString(): string {
    return this.model.versionString;
  }
  
  /**
   * Compare with another version
   */
  compareTo(other: Version): number {
    const otherModel = (other as DefaultVersion).model;
    
    if (this.model.major !== otherModel.major) return this.model.major - otherModel.major;
    if (this.model.minor !== otherModel.minor) return this.model.minor - otherModel.minor;
    if (this.model.patch !== otherModel.patch) return this.model.patch - otherModel.patch;
    return this.model.revision - otherModel.revision;
  }
  
  /**
   * Promote to next major version (X.0.0.0)
   * Immutable: Returns NEW component instance
   */
  async promoteMajor(): Promise<Version> {
    const newVersion = new DefaultVersion().init();
    newVersion.model.major = this.model.major + 1;
    newVersion.model.minor = 0;
    newVersion.model.patch = 0;
    newVersion.model.revision = 0;
    newVersion.model.versionString = `${newVersion.model.major}.0.0.0`;
    return newVersion;
  }
  
  /**
   * Promote to next minor version (X.Y.0.0)
   */
  async promoteMinor(): Promise<Version> {
    const newVersion = new DefaultVersion().init();
    newVersion.model.major = this.model.major;
    newVersion.model.minor = this.model.minor + 1;
    newVersion.model.patch = 0;
    newVersion.model.revision = 0;
    newVersion.model.versionString = `${newVersion.model.major}.${newVersion.model.minor}.0.0`;
    return newVersion;
  }
  
  /**
   * Promote to next patch version (X.Y.Z.0)
   */
  async promotePatch(): Promise<Version> {
    const newVersion = new DefaultVersion().init();
    newVersion.model.major = this.model.major;
    newVersion.model.minor = this.model.minor;
    newVersion.model.patch = this.model.patch + 1;
    newVersion.model.revision = 0;
    newVersion.model.versionString = `${newVersion.model.major}.${newVersion.model.minor}.${newVersion.model.patch}.0`;
    return newVersion;
  }
  
  /**
   * Promote to next revision (X.Y.Z.W)
   */
  async promoteRevision(): Promise<Version> {
    const newVersion = new DefaultVersion().init();
    newVersion.model.major = this.model.major;
    newVersion.model.minor = this.model.minor;
    newVersion.model.patch = this.model.patch;
    newVersion.model.revision = this.model.revision + 1;
    newVersion.model.versionString = `${newVersion.model.major}.${newVersion.model.minor}.${newVersion.model.patch}.${newVersion.model.revision}`;
    return newVersion;
  }
}
```
