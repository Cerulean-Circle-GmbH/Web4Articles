**2. Implemented findProjectRootFrom() Method**
```typescript
private findProjectRootFrom(startDir: string): string {
  let currentDir = path.resolve(startDir);
  
  // First priority: package.json + components/ directory
  // This is MORE SPECIFIC and correctly identifies test/data or project root
  while (currentDir !== path.dirname(currentDir)) {
    if (existsSync(path.join(currentDir, 'package.json')) &&
        existsSync(path.join(currentDir, 'components'))) {
      return currentDir;
    }
    currentDir = path.dirname(currentDir);
  }
  
  // Fallback: Look for .git
  currentDir = path.resolve(startDir);
  while (currentDir !== path.dirname(currentDir)) {
    if (existsSync(path.join(currentDir, '.git'))) {
      return currentDir;
    }
    currentDir = path.dirname(currentDir);
  }
  
  return path.resolve(startDir);
}
```
