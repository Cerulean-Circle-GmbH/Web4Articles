### **Error Category 1: Component Interface Method Mismatches**

**Problem:** Server components (HttpServer, WsServer, P2PServer) don't implement required Component interface methods

**AS IS - Component.interface.ts (Expected):**
```typescript
export interface Component {
  init(scenario: Scenario): this;
  start(): Promise<void>;        // ❌ Missing in server components
  stop(): Promise<void>;         // ❌ Missing in server components
  getIOR(): IOR;                 // ❌ Missing in server components
  isRunning(): boolean;          // ❌ Missing in server components
  toScenario(): Promise<Scenario>;
}
```

**AS IS - DefaultHttpServer.ts (Actual):**
```typescript
export class DefaultHttpServer implements HttpServer {
  // Has these methods:
  async startServer(): Promise<void> { ... }    // ❌ Named startServer not start
  async stopServer(): Promise<void> { ... }     // ❌ Named stopServer not stop
  isRunning(): boolean { ... }                  // ✅ Exists
  // Missing:
  // - start(): Promise<void>
  // - stop(): Promise<void>  
  // - getIOR(): IOR
}
```

**Compilation Error:**
```
src/ts/layer2/DefaultONCE.ts(307,9): error TS2739: Type 'DefaultHttpServer' is missing the following properties from type 'Component': start, stop, getIOR, isRunning
```

**SHOULD BE - Solution Options:**

**Option A: Add Wrapper Methods**
```typescript
export class DefaultHttpServer implements HttpServer {
  // Existing methods
  async startServer(): Promise<void> { ... }
  async stopServer(): Promise<void> { ... }
  
  // Add Component interface wrappers
  async start(): Promise<void> {
    return this.startServer();
  }
  
  async stop(): Promise<void> {
    return this.stopServer();
  }
  
  getIOR(): IOR {
    return this.iorComponent.toJSON();
  }
}
```

**Option B: Rename Methods**
```typescript
export class DefaultHttpServer implements HttpServer {
  // Rename to match Component interface
  async start(): Promise<void> {  // ✅ Matches Component interface
    this.data.state = 'starting';
    // ... implementation
  }
  
  async stop(): Promise<void> {   // ✅ Matches Component interface
    this.data.state = 'stopping';
    // ... implementation
  }
  
  getIOR(): IOR {
    return this.iorComponent.toJSON();
  }
}
```


**AS IS - Scenario Interface (Current):**
```typescript
export interface Scenario<T extends Model = Model> {
```

```typescript
export interface ONCEModel {

  loadedComponents: IOR[];
```

```typescript
// Add missing fields to ONCEModel
export interface ONCEModel extends Model {  // Explicitly extend Model
  uuid: string;
```

### **Error Category 4: Missing DefaultCLI.createForComponent() Method**

**Problem:** Old ONCECLI.ts calls createForComponent() static method that doesn't exist

**AS IS - ONCECLI.ts (Old Code):**
```typescript
class ONCECLI {
  private cli: DefaultCLI;
  private once: DefaultONCE;

  constructor() {
    this.once = new DefaultONCE();
    this.cli = DefaultCLI.createForComponent(this.once, 'ONCE');  // ❌ Method doesn't exist
  }
}
```

**Compilation Error:**
```
src/ts/layer5/ONCECLI.ts(23,27): error TS2339: Property 'createForComponent' does not exist on type 'typeof DefaultCLI'.
```

**SHOULD BE - Use Auto-Discovery Pattern:**
```typescript
import { DefaultONCE } from '../layer2/DefaultONCE.js';

class ONCECLI extends DefaultCLI {
  static async start(args: string[]): Promise<void> {
    const componentClass = DefaultONCE;
    const cli = new ONCECLI();
    cli.initWithComponentClass(componentClass, 'ONCE', '0.3.1.1');
    await cli.execute(args);
  }

  async execute(args: string[]): Promise<void> {
    if (args.length === 0 || args[0] === 'help' || args[0] === '--help') {
      this.showUsage();
      return;
    }

    const command = args[0];
    const commandArgs = args.slice(1);

    // Try dynamic command execution
    const executed = await this.executeDynamicCommand(command, commandArgs);
    if (!executed) {
      throw new Error(`Unknown command: ${command}`);
    }
  }

  showUsage(): void {
    this.generateDynamicUsage('once', '0.3.1.1');
  }
}

// Start the CLI
ONCECLI.start(process.argv.slice(2)).catch((error: Error) => {
  console.error(error.message);
  process.exit(1);
});
```
