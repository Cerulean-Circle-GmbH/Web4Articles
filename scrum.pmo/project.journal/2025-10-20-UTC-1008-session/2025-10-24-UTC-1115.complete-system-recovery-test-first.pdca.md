4. Result: Double headers, no shebang on line 1

describe('TC19: System-Wide Bash Executable Recovery', () => {

      // Count headers
      const blockHeaders = (content.match(/^\/\*\*/gm) || []).length;

      // Should have AT MOST one of each type

    // Verify shebang still on line 1

    expect(result.split('\n')[0]).toBe('#!/bin/bash');

**Phase 3: Implement Complete Fix**

```bash
# Fix ALL component executables, not just pdca/licensetool
find components -type f -executable ! -path "*/node_modules/*" | while read f; do
  if head -1 "$f" 2>/dev/null | grep -q "^/\*\*"; then
    node -e "
      const fs = require('fs');
      const content = fs.readFileSync('$f', 'utf-8');
      
      // Remove /** block
      let fixed = content.replace(/^\/\*\*[\s\S]*?\*\/\n+/, '');
      
      // If has # header but no shebang, add it
      if (fixed.startsWith('# SPDX-License') && !fixed.includes('#!/bin/bash')) {
        fixed = '#!/bin/bash\n' + fixed;
      }
      
      fs.writeFileSync('$f', fixed);
    " && echo "Fixed: $f"
  fi
done
```

async insertHeaderInternal(content: string, header: string): Promise<string> {
  // Extract shebang if present
  let shebang = '';
  let restOfContent = content;
  
  if (content.startsWith('#!')) {
    const firstNewline = content.indexOf('\n');
    shebang = content.substring(0, firstNewline + 1);
    restOfContent = content.substring(firstNewline + 1);
  }
  
  // Remove any existing headers from rest of content
  // ... existing header removal logic ...
  
  // Return: shebang + header + rest
  return shebang + header + '\n\n' + restOfContent.trimStart();
}
