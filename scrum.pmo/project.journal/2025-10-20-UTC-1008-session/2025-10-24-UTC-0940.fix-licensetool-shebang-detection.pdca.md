## 🔄 DO

### Proposed Fix

**Strategy:** Detect shebang BEFORE defaulting to 'block'

**Implementation:**

```typescript
async getCommentStyleInternal(filePath: string): Promise<'block' | 'hash' | 'html'> {
  const path = await import('path');
  const fs = await import('fs/promises');
  const ext = path.extname(filePath).toLowerCase();
  
  // Extension-based detection (fast path)
  if (['.ts', '.js', '.tsx', '.jsx', '.java', '.c', '.cpp', '.cs'].includes(ext)) {
    return 'block';
  }
  
  if (['.py', '.sh', '.yml', '.yaml'].includes(ext)) {
    return 'hash';
  }
  
  if (['.md', '.html', '.xml'].includes(ext)) {
    return 'html';
  }
  
  // NEW: Content-based detection for files without extension
  if (!ext) {
    try {
      const content = await fs.readFile(filePath, 'utf-8');
      const firstLine = content.split('\n')[0].trim();
      
      // Detect shebang
      if (firstLine.startsWith('#!')) {
        return 'hash'; // Bash, Python, Ruby, Perl scripts
      }
    } catch (err) {
      // If can't read file, fall through to default
    }
  }
  
  return 'block'; // Default for unknown types
}
```
