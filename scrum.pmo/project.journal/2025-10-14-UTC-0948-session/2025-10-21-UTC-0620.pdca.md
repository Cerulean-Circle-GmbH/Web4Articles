**Simple completion output:**

```typescript
async completion(what: string = '', filter: string = ''): Promise<this> {
  if (what === 'method' || what === '') {
    const methods = this.getAvailableMethods(filter);
    const methodNames = methods.map(m => m.name);
    
    // Detect single match
    if (methodNames.length === 1) {
      console.log(`DISPLAY: ✅ Single match found: ${methodNames[0]}`);
      console.log(`TRIGGER: __SINGLE__:${methodNames[0]}`);
      return this;
    }
    
    // Multiple matches - show hierarchical display
    if (methodNames.length > 0) {
      const formattedLines = this.formatMethodsHierarchicalLines(methods, filter);
      // Output each display line with DISPLAY: prefix
      formattedLines.forEach(line => {
        console.log(`DISPLAY: ${line}`);
      });
      
      // Output each word with WORD: prefix
      methodNames.forEach(name => {
        console.log(`WORD: ${name}`);
      });
    }
  } else if (what === 'parameter') {
    const values = this.getParameterValues(filter);
    
    // Check if callback needed
    if (this.needsCallback(filter)) {
      console.log(`TRIGGER: __CALLBACK__:${filter}ParameterCompletion`);
      return this;
    }
    
    if (values.length === 1) {
      console.log(`TRIGGER: __SINGLE__:${values[0]}`);
      return this;
    }
    
    // Output parameter values
    values.forEach(value => {
      console.log(`WORD: ${value}`);
    });
  }
  
  return this;
}

// Helper: Format methods as array of lines (no ANSI in TypeScript!)
private formatMethodsHierarchicalLines(methods: any[], filter: string): string[] {
  const lines: string[] = [];
  
  if (filter) {
    lines.push(`🔍 Filtered methods (${methods.length}):`);
  } else {
    lines.push(`📋 Available methods (${methods.length}):`);
  }
  
  methods.forEach((method, index) => {
    const signature = this.formatMethodSignature(method);
    lines.push(`  ${index + 1}: ${signature}`);
  });
  
  return lines;
}
```

**Key Points:**
- ✅ No flags (--lines, --display, etc.)
- ✅ No markers ([DISPLAY], [/DISPLAY])
- ✅ Simple prefixes (DISPLAY:, WORD:, TRIGGER:)
- ✅ One grep line extracts what bash needs
- ✅ TypeScript decides format (method controls output)

### **bash Implementation**

**Entire completion function (~20 lines!):**

```bash
_web4tscomponent() {
    local cur prev cmd result words trigger display
    COMPREPLY=()
    cur="${COMP_WORDS[COMP_CWORD]}"
    prev="${COMP_WORDS[COMP_CWORD-1]}"
    cmd="${COMP_WORDS[0]}"
    
    # Get completion from TypeScript
    result=$(timeout 30s "$cmd" completion method "$cur" 2>/dev/null)
    
    # Handle timeout
    [ $? -eq 124 ] && { printf "\n⏱️  Timeout\n" >&2; return 0; }
    
    # Extract display lines (show to user, then continue)
    display=$(echo "$result" | grep "^DISPLAY: " | cut -d' ' -f2-)
    [ -n "$display" ] && printf "\n%s\n\n" "$display"
    
    # Check for triggers
    trigger=$(echo "$result" | grep "^TRIGGER: " | cut -d' ' -f2-)
    if [[ "$trigger" == __SINGLE__:* ]]; then
        COMPREPLY=("${trigger#__SINGLE__:} ")
        return 0
    fi
    if [[ "$trigger" == __CALLBACK__:* ]]; then
        # Recursive call to completeParameter...
        local callback="${trigger#__CALLBACK__:}"
        result=$(timeout 30s "$cmd" completeParameter "$callback" "$prev" "$cur" 2>/dev/null)
        # Process result again...
    fi
    
    # Extract words for compgen
    words=$(echo "$result" | grep "^WORD: " | cut -d' ' -f2-)
    COMPREPLY=($(compgen -W "$words" -- "$cur"))
}

complete -F _web4tscomponent web4tscomponent
```
