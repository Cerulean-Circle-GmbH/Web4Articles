### **Simple bash Implementation**

**Entire completion function (~30 lines, not 200!):**

```bash
_web4tscomponent() {
    local cur prev cmd
    COMPREPLY=()
    cur="${COMP_WORDS[COMP_CWORD]}"
    prev="${COMP_WORDS[COMP_CWORD-1]}"
    cmd="${COMP_WORDS[0]}"
    
    # Get completion from TypeScript (with display mode for hierarchical)
    local result=$(timeout 30s "$cmd" completion method "$cur" --display 2>/dev/null)
    local exit_code=$?
    
    # Handle timeout
    if [ $exit_code -eq 124 ]; then
        printf "\n⏱️  Timeout - completion cancelled\n" >&2
        return 0
    fi
    
    # Extract display section (between [DISPLAY] and [/DISPLAY])
    if echo "$result" | grep -q "\[DISPLAY\]"; then
        local display=$(echo "$result" | sed -n '/\[DISPLAY\]/,/\[\/DISPLAY\]/p' | sed '1d;$d')
        if [ -n "$display" ]; then
            printf "\n%s\n\n" "$display"
        fi
    fi
    
    # Extract words section (between [WORDS] and [/WORDS])
    local words=""
    if echo "$result" | grep -q "\[WORDS\]"; then
        words=$(echo "$result" | sed -n '/\[WORDS\]/,/\[\/WORDS\]/p' | sed '1d;$d')
    else
        # No markers - assume entire output is words
        words="$result"
    fi
    
    # Check for special triggers
    if [[ "$words" == __CALLBACK__:* ]]; then
        local callback_name="${words#__CALLBACK__:}"
        # Execute callback
        result=$(timeout 30s "$cmd" completeParameter "$callback_name" "$prev" "$cur" --display 2>/dev/null)
        # Recursive processing...
        words=$(echo "$result" | sed -n '/\[WORDS\]/,/\[\/WORDS\]/p' | sed '1d;$d')
    fi
    
    if [[ "$words" == __SINGLE__:* ]]; then
        local single_match="${words#__SINGLE__:}"
        COMPREPLY=("$single_match ")
        return 0
    fi
    
    # Simple compgen with words
    COMPREPLY=($(compgen -W "$words" -- "$cur"))
}

complete -F _web4tscomponent web4tscomponent
```

**That's it. ~30 lines. TypeScript does EVERYTHING else.**

### **TypeScript Changes**

**File:** `components/Web4TSComponent/0.3.14.2/src/ts/layer2/DefaultCLI.ts`

**Add output mode parameter:**

```typescript
async completion(
  what: string = '',
  filter: string = '',
  mode: 'words' | 'lines' | 'display' = 'words'
): Promise<this> {
  if (what === 'method' || what === '') {
    const methods = this.getAvailableMethods(filter);
    const methodNames = methods.map(m => m.name);
    
    // Detect single match
    if (methodNames.length === 1) {
      console.log(`__SINGLE__:${methodNames[0]}`);
      return this;
    }
    
    // Output based on mode
    if (mode === 'display') {
      // Hierarchical display + words
      console.log('[DISPLAY]');
      console.log(this.formatMethodsHierarchical(methods, filter));
      console.log('[/DISPLAY]');
      console.log('[WORDS]');
      console.log(methodNames.join(' '));
      console.log('[/WORDS]');
    } else if (mode === 'lines') {
      // One per line
      console.log(methodNames.join('\n'));
    } else {
      // Space-separated (default)
      console.log(methodNames.join(' '));
    }
  } else if (what === 'parameter') {
    // Similar for parameters
    const values = this.getParameterValues(filter);
    
    // Check if callback needed
    if (this.needsCallback(filter)) {
      console.log(`__CALLBACK__:${filter}ParameterCompletion`);
      return this;
    }
    
    if (values.length === 1) {
      console.log(`__SINGLE__:${values[0]}`);
      return this;
    }
    
    if (mode === 'lines') {
      console.log(values.join('\n'));
    } else {
      console.log(values.join(' '));
    }
  }
  
  return this;
}
```

- ✅ Display formatting in TypeScript (returns `[DISPLAY]...[/DISPLAY]`)
- ✅ Simple word output (space or line separated)
