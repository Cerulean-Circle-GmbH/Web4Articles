```bash
_web4_tscompletion() {
    local cur prev cmd result words display word_count
    COMPREPLY=()
    cur="${COMP_WORDS[COMP_CWORD]}"
    prev="${COMP_WORDS[COMP_CWORD-1]}"
    cmd="${COMP_WORDS[0]}"
    
    # ... existing code ...
    
    # NEW: Extract DISPLAY lines (show to user)
    display=$(echo "$result" | grep "^DISPLAY: " | cut -d' ' -f2-)
    [ -n "$display" ] && printf "\n%s\n\n" "$display"
    
    # NEW: Extract WORD lines (for compgen)
    words=$(echo "$result" | grep "^WORD: " | cut -d' ' -f2-)
    
    # NEW: Single word auto-complete detection
    if [ -n "$words" ]; then
        word_count=$(echo "$words" | wc -w | tr -d ' ')
        if [ "$word_count" -eq 1 ]; then
            COMPREPLY=("$words ")
            return 0
        fi
        
        # Multiple words - use compgen
        COMPREPLY=($(compgen -W "$words" -- "$cur"))
    else
        # Fallback to old behavior (for non-migrated methods)
        COMPREPLY=($(compgen -W "$result" -- "$cur"))
    fi
}
```

```typescript
    const values = await (this as any)[callbackName](contextArgs);
    
    // NEW: Output with WORD: prefix
```
