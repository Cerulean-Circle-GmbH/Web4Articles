```typescript
async completion(what: string = '', filter: string = '', jsonOutput: boolean = false): Promise<this> {
  const result = this.generateCompletionResult(what, filter);
  
  if (jsonOutput) {
    // Output pure JSON (no colors, no extra text)
    console.log(JSON.stringify(result));
  } else {
    // Current behavior: formatted output
    if (result.display) {
      console.log(result.display);
    }
  }
  
  return this;
}

private generateCompletionResult(what: string, filter: string): CompletionResult {
  if (what === 'method' || what === '') {
    const methods = this.getAvailableMethods(filter);
    const formatted = this.formatMethodsHierarchical(methods, filter);
    
    // Detect single match (no ANSI parsing needed - we have raw data!)
    const singleMatch = methods.length === 1 ? methods[0].name : null;
    
    return {
      format: singleMatch ? 'auto-complete' : 'hierarchical',
      display: formatted,
      completions: methods.map(m => m.name),
      single_match: singleMatch,
      error: null
    };
  }
  
  if (what === 'parameter') {
    // Similar logic for parameter completion
    const values = this.getParameterValues(filter);
    const singleMatch = values.length === 1 ? values[0] : null;
    
    return {
      format: singleMatch ? 'auto-complete' : 'compgen',
      display: values.join('\n'),
      completions: values,
      single_match: singleMatch,
      error: null
    };
  }
  
  return {
    format: 'error',
    display: '',
    completions: [],
    single_match: null,
    error: `Unknown completion type: ${what}`
  };
}
```

```bash
# JSON parsing helper - extract field from JSON
parse_json_field() {
    local json="$1"
    local field="$2"
    
    # Use grep + sed for simple JSON parsing (no jq dependency!)
    echo "$json" | grep -o "\"$field\":[^,}]*" | sed 's/"[^"]*"://; s/[",]//g; s/^ *//; s/ *$//'
}

# JSON array parsing helper - extract array values
parse_json_array() {
    local json="$1"
    local field="$2"
    
    echo "$json" | grep -o "\"$field\":\[[^\]]*\]" | sed 's/"[^"]*"://; s/\[//; s/\]//; s/"//g; s/,/ /g'
}
```

**Refactor completion functions to use JSON:**

```bash
# Method completion (replaces lines 50-252)
_web4tscomponent_method_completion() {
    local cmd="$1"
    local filter="$2"
    
    # Get JSON result from TypeScript
    local json_result=$(timeout 30s "$cmd" completion method "$filter" --json 2>/dev/null)
    local exit_code=$?
    
    # Handle timeout
    if [ $exit_code -eq 124 ]; then
        printf "\n${RED}⏱️  TIMEOUT: Completion took >30s${NO_COLOR}\n" >&2
        printf "   Your terminal is ready. Try a simpler completion.\n\n" >&2
        COMPREPLY=()
        return 0
    fi
    
    # Parse JSON
    local format=$(parse_json_field "$json_result" "format")
    local display=$(parse_json_field "$json_result" "display")
    local completions=$(parse_json_array "$json_result" "completions")
    local single_match=$(parse_json_field "$json_result" "single_match")
    local error=$(parse_json_field "$json_result" "error")
    
    # Handle error
    if [ "$format" = "error" ]; then
        printf "\n${RED}ERROR: $error${NO_COLOR}\n" >&2
        COMPREPLY=()
        return 0
    fi
    
    # Handle auto-complete (single match)
    if [ "$format" = "auto-complete" ] && [ -n "$single_match" ]; then
        # Show hierarchical display
        if [ -n "$display" ]; then
            printf "\n${CYAN}🔍 Single match found:${NO_COLOR}\n"
            echo "$display"
            printf "\n"
        fi
        # Auto-complete the single match
        COMPREPLY=("$single_match ")
        return 0
    fi
    
    # Handle hierarchical display
    if [ "$format" = "hierarchical" ] || [ "$format" = "both" ]; then
        if [ -n "$display" ]; then
            printf "\n"
            echo "$display"
            printf "\n"
        fi
    fi
    
    # Handle compgen completions
    if [ -n "$completions" ]; then
        COMPREPLY=($(compgen -W "$completions" -- "$filter"))
    else
        COMPREPLY=()
    fi
}
```
