**Added `formatCompletionOutput()` helper (lines 1295-1330):**

```typescript
private formatCompletionOutput(values: string[]): void {
  const lines: string[] = [];
  
  // Detect complex format (numbered lines like "1: methodName <params>")
  const hasNumberedRefs = values.some((v: string) => v.match(/^\d+:/));
  const hasSpaces = values.some((v: string) => v.includes(' '));
  
  if (hasNumberedRefs || hasSpaces) {
    // Complex format: Add DISPLAY lines
    values.forEach((line: string) => {
      lines.push(`DISPLAY: ${line}`);
    });
    
    // Extract words and add WORD lines
    values.forEach((line: string) => {
      const match = line.match(/^\d+:\s*(\S+)/);
      const word = match ? match[1] : line.split(' ')[0];
      lines.push(`WORD: ${word}`);
    });
  } else {
    // Simple format: Just WORD lines
    values.forEach((word: string) => {
      lines.push(`WORD: ${word}`);
    });
  }
  
  // ONE console.log for entire block (efficient!)
  console.log(lines.join('\n'));
}
```

### Bash Implementation

**NEW protocol parsing (source.env lines 411-496):**

```bash
# NEW PROTOCOL: Extract DISPLAY and WORD lines (DRY: one parsing, multiple uses)
local display_lines=()
local word_values=()

# Parse output for DISPLAY: and WORD: prefixes
while IFS= read -r line; do
    if [[ "$line" =~ ^DISPLAY:[[:space:]]*(.*) ]]; then
        display_lines+=("${BASH_REMATCH[1]}")
    elif [[ "$line" =~ ^WORD:[[:space:]]*(.*) ]]; then
        word_values+=("${BASH_REMATCH[1]}")
    fi
done <<< "$out"

# Smart Hybrid Detection (OOSH-inspired pattern)
if [ ${#display_lines[@]} -gt 0 ]; then
    # Multi-LINE: Show DISPLAY, use WORD for COMPREPLY
    printf "\n"
    for line in "${display_lines[@]}"; do
        printf "%s\n" "$line"
    done
    
    local tokens=()
    for word in "${word_values[@]}"; do
        tokens+=("$word ")
    done
    COMPREPLY=("${tokens[@]}")
    return 0
    
elif [ ${#word_values[@]} -gt 0 ]; then
    # Single-word: Use compgen with WORD values
    local words_string="${word_values[*]}"
    COMPREPLY=( $(compgen -W "$words_string" -- "$cur") )
    
else
    # FALLBACK: Backward compatible (no protocol)
    COMPREPLY=( $(compgen -W "$out" -- "$cur") )
fi
```
