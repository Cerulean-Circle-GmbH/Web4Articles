```typescript:4680-4751:components/Web4TSComponent/0.3.13.2/src/ts/layer2/DefaultWeb4TSComponent.ts
async setCICDVersion(
  componentName: string = 'current',
  version: string = 'current'
): Promise<this> {
  // Resolve parameters
  let targetComponent: string;
  let targetVersion: string;
  
  if (componentName === 'current' || version === 'current') {
    const context = this.getComponentContext();
    if (!context) {
      throw new Error('No component context. Use "on <component> <version>" first or provide explicit parameters.');
    }
    targetComponent = componentName === 'current' ? context.component : componentName;
    targetVersion = version === 'current' ? context.version : version;
  } else {
    targetComponent = componentName;
    targetVersion = version;
  }
  
  // Parse version for build number logic
  const versionParts = targetVersion.split('.');
  const buildNumber = parseInt(versionParts[3] || '0', 10);
  const isBuild0 = buildNumber === 0;
  
  console.log(`🔗 Setting up CI/CD link infrastructure for ${targetComponent} ${targetVersion}`);
  console.log(`   Build type: ${isBuild0 ? 'Build 0 (Production)' : `Build ${buildNumber} (Development/Test)`}`);
  
  const fs = await import('fs/promises');
  const componentDir = this.resolveComponentDirectory(targetComponent);
  
  // Set latest symlink (always)
  const latestLink = path.join(componentDir, 'latest');
  try {
    await fs.unlink(latestLink).catch(() => {});
    await fs.symlink(targetVersion, latestLink);
    console.log(`   ✅ latest → ${targetVersion}`);
  } catch (error) {
    console.log(`   ⚠️  Could not set latest link: ${error}`);
  }
  
  if (isBuild0) {
    // Build 0: Set as prod
    const prodLink = path.join(componentDir, 'prod');
    try {
      await fs.unlink(prodLink).catch(() => {});
      await fs.symlink(targetVersion, prodLink);
      console.log(`   ✅ prod → ${targetVersion} (stable build 0)`);
    } catch (error) {
      console.log(`   ⚠️  Could not set prod link: ${error}`);
    }
  } else {
    // Build 1+: Set as dev and test
    const devLink = path.join(componentDir, 'dev');
    const testLink = path.join(componentDir, 'test');
    
    try {
      await fs.unlink(devLink).catch(() => {});
      await fs.symlink(targetVersion, devLink);
      console.log(`   ✅ dev → ${targetVersion} (build ${buildNumber} development)`);
    } catch (error) {
      console.log(`   ⚠️  Could not set dev link: ${error}`);
    }
    
    try {
      await fs.unlink(testLink).catch(() => {});
      await fs.symlink(targetVersion, testLink);
      console.log(`   ✅ test → ${targetVersion} (build ${buildNumber} testing)`);
    } catch (error) {
      console.log(`   ⚠️  Could not set test link: ${error}`);
    }
  }
  
  console.log(`✅ CI/CD links established for ${targetComponent} ${targetVersion}`);
  return this;
}
```
