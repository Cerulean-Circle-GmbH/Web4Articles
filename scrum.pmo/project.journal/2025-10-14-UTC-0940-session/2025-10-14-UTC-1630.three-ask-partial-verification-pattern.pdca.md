tested=$actual_count

#!/bin/bash
# verify_all_links.sh - No partial verification possible
# Usage: ./verify_all_links.sh path/to/file.md

file="$1"
cd "$(dirname "$file")"
links=$(grep -o '\](.*\.md)' "$file" | sed 's/](\(.*\))/\1/')
total=$(echo "$links" | wc -l)
broken=0

echo "Verifying $total links in $(basename "$file")..."

    exit 0
else
